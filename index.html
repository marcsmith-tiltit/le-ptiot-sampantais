export default function NewspaperReaderWebsite() {

  const [selectedLanguage, setSelectedLanguage] = useState("fr-FR");

  const [selectedArticleId, setSelectedArticleId] = useState(articles[0].id);

  const [isReading, setIsReading] = useState(false);

  const audioRef = useRef(null);


  const selectedArticle = useMemo(

    () => articles.find((article) => article.id === selectedArticleId) || articles[0],

    [selectedArticleId]

  );


  const translatedText = selectedArticle.text[selectedLanguage] || selectedArticle.text["en-GB"];


  function stopReading() {

    if (audioRef.current) {

      audioRef.current.pause();

      audioRef.current.currentTime = 0;

    }

    setIsReading(false);

  }


  function readArticle(article = selectedArticle) {

    stopReading();

    const audioPath = article.audio[selectedLanguage] || article.audio["en-GB"];

    const audio = new Audio(audioPath);

    audioRef.current = audio;

    audio.onended = () => setIsReading(false);

    audio.onerror = () => setIsReading(false);

    setIsReading(true);

    audio.play().catch(() => setIsReading(false));

  }


  function chooseArticle(article) {

    setSelectedArticleId(article.id);

    readArticle(article);

  }


  return (

    <main className="min-h-screen bg-neutral-50 p-4 text-neutral-900 md:p-8">

      <div className="mx-auto max-w-7xl">

        <motion.header

          initial={{ opacity: 0, y: -10 }}

          animate={{ opacity: 1, y: 0 }}

          className="mb-6 flex flex-col gap-4 md:flex-row md:items-end md:justify-between"

        >

          <div>

            <div className="mb-2 flex items-center gap-2 text-sm font-medium uppercase tracking-[0.25em] text-neutral-500">

              <Newspaper className="h-4 w-4" /> Interactive newspaper reader

            </div>

            <h1 className="text-3xl font-semibold tracking-tight md:text-5xl">

              Click an article. Hear it in your language.

            </h1>

            <p className="mt-3 max-w-2xl text-base text-neutral-600">

              Page 1 is set up with clickable zones. Put the uploaded image in the public folder as page-1-la-une-v2.png, then add the MP3 files for each article and language.

            </p>

          </div>


          <Card className="rounded-2xl shadow-sm">

            <CardContent className="flex items-center gap-3 p-4">

              <Languages className="h-5 w-5 text-neutral-500" />

              <label className="text-sm font-medium" htmlFor="language">

                Language

              </label>

              <select

                id="language"

                value={selectedLanguage}

                onChange={(event) => setSelectedLanguage(event.target.value)}

                className="rounded-xl border border-neutral-300 bg-white px-3 py-2 text-sm shadow-sm outline-none focus:ring-2 focus:ring-neutral-300"

              >

                {LANGUAGES.map((language) => (

                  <option key={language.code} value={language.code}>

                    {language.label}

                  </option>

                ))}

              </select>

            </CardContent>

          </Card>

        </motion.header>


        <div className="grid gap-6 lg:grid-cols-[1.35fr_0.65fr]">

          <Card className="overflow-hidden rounded-2xl shadow-sm">

            <CardContent className="p-4 md:p-6">

              <div className="mb-4 flex items-center gap-2 text-sm text-neutral-600">

                <MousePointerClick className="h-4 w-4" /> Click a highlighted area on the newspaper.

              </div>


              <div className="relative aspect-[3/4] w-full overflow-hidden rounded-2xl border border-neutral-200 bg-white shadow-sm">

                {/* Replace NewspaperPlaceholder with an image, for example:

                    <img src="/newspaper.jpg" alt="Scanned newspaper" className="h-full w-full object-cover" />

                */}

                <img

                  src="/page-1-la-une-v2.png"

                  alt="Page 1 - La Une du P’tiot Sampantais"

                  className="h-full w-full object-cover"

                />


                {articles.map((article) => {

                  const isActive = article.id === selectedArticleId;

                  return (

                    <button

                      key={article.id}

                      type="button"

                      onClick={() => chooseArticle(article)}

                      aria-label={`Read ${article.title}`}

                      className={`absolute rounded-xl border-2 transition-all duration-200 focus:outline-none focus:ring-4 focus:ring-neutral-500/20 ${

                      <span className="absolute left-2 top-2 rounded-full bg-neutral-900 px-2 py-1 text-xs font-medium text-white shadow-sm">

                        {article.title}

                      </span>

                    </button>

                  );

                })}

              </div>

            </CardContent>

          </Card>


          <aside className="space-y-6">

            <Card className="rounded-2xl shadow-sm">

              <CardContent className="p-6">

                <p className="mb-2 text-sm font-medium uppercase tracking-[0.2em] text-neutral-500">

                  Selected article

                </p>

                <h2 className="mb-4 text-2xl font-semibold">{selectedArticle.title}</h2>

                <p className="min-h-40 rounded-2xl bg-neutral-100 p-4 text-lg leading-relaxed text-neutral-800">

                  {translatedText}

                </p>


                <div className="mt-5 flex flex-wrap gap-3">

                  <Button onClick={() => readArticle()} className="rounded-2xl px-5">

                    <Volume2 className="mr-2 h-4 w-4" /> Read aloud

                  </Button>

                  <Button variant="outline" onClick={stopReading} className="rounded-2xl px-5">

                    <Square className="mr-2 h-4 w-4" /> Stop

                  </Button>

                </div>

                {isReading && (

                  <p className="mt-3 text-sm text-neutral-500">Reading aloud…</p>

                )}

              </CardContent>

            </Card>


            <Card className="rounded-2xl shadow-sm">

              <CardContent className="p-6">

                <h3 className="mb-3 text-lg font-semibold">How to finish it</h3>

                <ol className="space-y-2 text-sm leading-relaxed text-neutral-600">

                  <li>1. Save this page image as <code>/public/page-1-la-une-v2.png</code>.</li>

                  <li>2. Keep the MP3 files in <code>/public/audio/</code>.</li>

                  <li>3. Fine-tune each clickable article zone if needed by adjusting <code>left</code>, <code>top</code>, <code>width</code>, and <code>height</code>.</li>

                  <li>4. Export one MP3 per article and per language from NaturalReaders.</li>

                  <li>5. Put the files in <code>/public/audio/</code>, using names such as <code>mayor-notice-fr.mp3</code>.</li>

                  <li>6. Replace the sample article text with real transcriptions and translations.</li>

                </ol>

              </CardContent>

            </Card>

          </aside>

        </div>

      </div>

    </main>

  );

}
