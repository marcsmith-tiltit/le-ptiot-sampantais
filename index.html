  <script>

    let currentPage = 1;


    function pageNumber(n) {

      return String(n).padStart(2, "0");

    }


    function selectArticle(articleId, title) {

      const language = document.getElementById("language").value;

      const audioPlayer = document.getElementById("audioPlayer");


      document.getElementById("articleTitle").textContent = title;


      const audioPath = `audio/page-${pageNumber(currentPage)}/${articleId}-${language}.mp3`;


      audioPlayer.src = audioPath;

      audioPlayer.play();


      document.getElementById("status").textContent =

        `Lecture : ${title} (${language})`;


      document.querySelectorAll(".article-zone").forEach(zone => {

        zone.classList.remove("active");

      });


      event.target.classList.add("active");

    }


    function updatePage() {

      document.getElementById("pageImage").src =

        `pages/page-${pageNumber(currentPage)}.png`;


      document.getElementById("pageImage").alt =

        `Page ${currentPage} du journal Le P’tiot Sampantais`;


      document.getElementById("pageLabel").textContent =

        `Page ${currentPage} / 12`;


      document.getElementById("articleTitle").textContent =

        "Choisissez un article";


      document.getElementById("status").textContent =

        "Cliquez sur une zone jaune de la page.";


      document.getElementById("audioPlayer").pause();

      document.getElementById("audioPlayer").removeAttribute("src");


      document.querySelectorAll(".article-zone").forEach(zone => {

        zone.classList.remove("active");

      });

    }


    function nextPage() {

      if (currentPage < 12) {

        currentPage++;

        updatePage();

      }

    }


    function previousPage() {

      if (currentPage > 1) {

        currentPage--;

        updatePage();

      }

    }
